import numpy as np

class vetor:
    tamanho = None

    def __init__(self, *valores):
        if len(valores) != self.tamanho:
            raise ValueError(f"Este vetor precisa de {self.tamanho} valores para ser criado.")
        
        self.valores = np.array(valores)

    def __str__(self):
        return f"{self.__class__.__name__}{tuple(self.valores.tolist())}"

    def soma(self, outro):
        return self.__class__(*(self.valores + outro.valores).tolist())

    def escalar(self, k):
        return self.__class__(*(self.valores * k).tolist())
